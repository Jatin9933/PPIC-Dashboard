// --- FIREBASE DATE PICKER & DATA FETCH INTEGRATION --- // 1. Fetch Data for a Specific Date selected from Firebase Cloud function fetchReportByDate() { const selectedDate = document.getElementById('firebaseDatePicker').value; if (!selectedDate) { alert("⚠️ Please select a date first!"); return; } if (!window.firebaseDb) { alert("❌ Firebase is not initialized."); return; } // Convert date format (e.g., YYYY-MM-DD) to match database path structure if required, // or fetch from a node structured by date: 'ppic_reports/history/' + selectedDate const datePath = `ppic_reports/history/${selectedDate}`; window.dbGet(window.dbChild(window.dbRef(window.firebaseDb), datePath)).then((snapshot) => { if (snapshot.exists()) { const val = snapshot.val(); if (val && val.data) { generatedReportRows = val.data; updateTableFilterDropdowns(); renderTable(); alert(`✅ Successfully loaded report for date: ${selectedDate}`); } else { alert(`⚠️ Data format invalid for date: ${selectedDate}`); } } else { alert(`❌ No report found in Firebase for date: ${selectedDate}. Falling back to 'latest'.`); loadFromFirebase(); // Fallback to latest if specific date doesn't exist } }).catch((error) => { console.error("Error fetching data by date:", error); alert("❌ Error fetching data: " + error.message); }); } // Update saveToFirebase to also save under historical date path for date-picker access function saveToFirebaseWithDate() { if (!window.firebaseDb) { alert("Firebase configuration is missing."); return; } if (generatedReportRows.length === 0) { alert("No data to save!"); return; } const todayDateStr = new Date().toISOString().slice(0, 10); // YYYY-MM-DD const payload = { lastUpdated: new Date().toISOString(), data: generatedReportRows }; // Save to both 'latest' and historical node based on current date const latestRef = window.dbRef(window.firebaseDb, 'ppic_reports/latest'); const historyRef = window.dbRef(window.firebaseDb, `ppic_reports/history/${todayDateStr}`); Promise.all([ window.dbSet(latestRef, payload), window.dbSet(historyRef, payload) ]).then(() => { alert("✅ Report saved successfully to Firebase Cloud (Latest & History)!"); }).catch((error) => { console.error("Firebase save error:", error); alert("❌ Firebase Error: " + error.message); }); }
